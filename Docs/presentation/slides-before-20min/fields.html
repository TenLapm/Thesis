public SimNode NextStep(SimNode node, MovementClass movement, bool raider) {
  SimNode next = raider && RaidGoals > 0 && node.RaidCost != Infinity
    ? NextRaid(node) : NextOf(node, movement);
  if (LureGoals > 0 && node.LureCost <= LureRadius * CardinalCost)
    next = NextLure(node);
  return next;
}

The agent pass and the route lines on screen both call NextStep, so a drawn line cannot show a way the enemies do not walk. Flyers use no field.
